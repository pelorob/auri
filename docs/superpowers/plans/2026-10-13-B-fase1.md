## Global Constraints

- Solo iPad 11ª gen: 1180×820 punti in orizzontale (barra laterale), 820×1180 in verticale (barra in basso sotto i 900 punti di larghezza).
- Colori: viola `#5B3DE6`, viola2 `#6B50F2`, viola3 `#4730C4`, violaTenue `#ECE8FF`, violaChip `#E4DEFF`, lime `#D6F35B`, inchiostro `#19142E`, grigio `#6B6785`, grigioChiaro `#8A87A0`, sfondo `#F2F1FA`.

- Testi in italiano, rivolti all'utente al femminile ("Iscritta", "Pronta").
- Numeri in formato italiano: virgola decimale, media con 2 decimali, base di laurea con 1 decimale, "—" se non disponibile.
- Default delle impostazioni: `cfuTotali` 180, `valoreLode` 30 (30–33), `bloccoAttivo` false, `minutiRiblocco` 5, preavvisi `7@9:00,1@9:00,0@7:30`.
- Notifiche: al massimo 60 programmate. ID notifica = `appelloId * 10 + indicePreavviso`, firma = 0.
- Avviso firma: si mostra con ≤ 3 giorni. Notifica il giorno prima della scadenza alle 9:00.
- Promemoria backup: se sono passati ≥ 7 giorni dall'ultimo backup (o dal primo avvio).
- Nessun server, nessuna rete richiesta.

```dart
class VoceLibretto {
  const VoceLibretto({required this.cfu, required this.superato, required this.aVoto, this.voto, this.lode = false});
  final int cfu;
  final bool superato;
  final bool aVoto;
  final int? voto;
  final bool lode;
}

class Riepilogo {
  const Riepilogo({required this.sommaPesata, required this.cfuConVoto, required this.numeroVoti, required this.sommaVoti, required this.cfuOttenuti});
  final double sommaPesata;
  final int cfuConVoto;
  final int numeroVoti;
  final double sommaVoti;
  final int cfuOttenuti;

  double? get mediaPonderata => cfuConVoto == 0 ? null : sommaPesata / cfuConVoto;
  double? get mediaAritmetica => numeroVoti == 0 ? null : sommaVoti / numeroVoti;
  double? get baseLaurea => mediaPonderata == null ? null : baseDaMedia(mediaPonderata!);
}

double valoreVoto(int voto, bool lode, int valoreLode) => voto == 30 && lode ? valoreLode.toDouble() : voto.toDouble();

double baseDaMedia(double media) => media * 110 / 30;

Riepilogo calcolaRiepilogo(Iterable<VoceLibretto> voci, {required int valoreLode}) {
  var sommaPesata = 0.0, sommaVoti = 0.0;
  var cfuConVoto = 0, numeroVoti = 0, cfuOttenuti = 0;
  for (final v in voci) {
    if (!v.superato) continue;
    cfuOttenuti += v.cfu;
    if (v.aVoto && v.voto != null) {
      final x = valoreVoto(v.voto!, v.lode, valoreLode);
      sommaPesata += x * v.cfu;
      cfuConVoto += v.cfu;
      sommaVoti += x;
      numeroVoti++;
    }
  }
  return Riepilogo(sommaPesata: sommaPesata, cfuConVoto: cfuConVoto, numeroVoti: numeroVoti, sommaVoti: sommaVoti, cfuOttenuti: cfuOttenuti);
}

double simulaMedia(Riepilogo r, {required int cfu, required int voto, bool lode = false, required int valoreLode}) =>
    (r.sommaPesata + valoreVoto(voto, lode, valoreLode) * cfu) / (r.cfuConVoto + cfu);

sealed class MediaNecessaria {
  const MediaNecessaria();
}

class Raggiungibile extends MediaNecessaria {
  const Raggiungibile(this.valore);
  final double valore;
}

class BastaPassare extends MediaNecessaria {
  const BastaPassare();
}

class FuoriPortata extends MediaNecessaria {
  const FuoriPortata();
}

MediaNecessaria? mediaNecessaria(Riepilogo r, {required int obiettivo, required int cfuTotali}) {
  final mancanti = cfuTotali - r.cfuOttenuti;
  if (mancanti <= 0) return null;
  final req = (obiettivo * 30 / 110 * (r.cfuConVoto + mancanti) - r.sommaPesata) / mancanti;
  if (req > 30) return const FuoriPortata();
  if (req < 18) return const BastaPassare();
  return Raggiungibile(req);
}
```

```dart
import 'formati.dart';
import 'modelli.dart';

const preavvisiPredefiniti = '7@9:00,1@9:00,0@7:30';

class Preavviso {
  const Preavviso(this.giorniPrima, this.ora, this.minuto);
  final int giorniPrima;
  final int ora;
  final int minuto;

  @override
  bool operator ==(Object o) => o is Preavviso && o.giorniPrima == giorniPrima && o.ora == ora && o.minuto == minuto;
  @override
  int get hashCode => Object.hash(giorniPrima, ora, minuto);
  @override
  String toString() => '$giorniPrima@$ora:${minuto.toString().padLeft(2, '0')}';
}

List<Preavviso> leggiPreavvisi(String s) {
  final out = <Preavviso>[];
  final re = RegExp(r'^(\d{1,2})@(\d{1,2}):(\d{2})$');
  for (final parte in s.split(',')) {
    final m = re.firstMatch(parte.trim());
    if (m == null) continue;
    final g = int.parse(m[1]!), h = int.parse(m[2]!), mi = int.parse(m[3]!);
    if (h > 23 || mi > 59) continue;
    out.add(Preavviso(g, h, mi));
  }
  return out;
}

String scriviPreavvisi(List<Preavviso> p) => p.map((x) => x.toString()).join(',');

class AppelloPerNotifica {
  const AppelloPerNotifica({required this.id, required this.corso, required this.dataOra, required this.tipo, this.aula, required this.preavvisi});
  final int id;
  final String corso;
  final DateTime dataOra;
  final TipoAppello tipo;
  final String? aula;
  final List<Preavviso> preavvisi;
}

class NotificaPianificata {
  const NotificaPianificata({required this.id, required this.quando, required this.titolo, required this.testo});
  final int id;
  final DateTime quando;
  final String titolo;
  final String testo;
}

String testoNotifica(AppelloPerNotifica a, int giorniPrima) {
  final quando = switch (giorniPrima) { 0 => 'oggi', 1 => 'domani', _ => 'tra $giorniPrima giorni' };
  final dettagli = [dataLunga(a.dataOra), oraMinuti(a.dataOra), if (a.aula != null && a.aula!.trim().isNotEmpty) a.aula!.trim()].join(', ');
  return '${a.corso}: ${a.tipo.etichetta} $quando ($dettagli)';
}

List<NotificaPianificata> pianificaNotifiche(Iterable<AppelloPerNotifica> appelli, DateTime ora, {int massimo = 60}) {
  final tutte = <NotificaPianificata>[];
  for (final a in appelli) {
    for (var i = 0; i < a.preavvisi.length && i < 10; i++) {
      final p = a.preavvisi[i];
      final giorno = DateTime(a.dataOra.year, a.dataOra.month, a.dataOra.day - p.giorniPrima);
      final quando = DateTime(giorno.year, giorno.month, giorno.day, p.ora, p.minuto);
      if (!quando.isAfter(ora)) continue;
      tutte.add(NotificaPianificata(id: a.id * 10 + i, quando: quando, titolo: 'auri', testo: testoNotifica(a, p.giorniPrima)));
    }
  }
  tutte.sort((x, y) => x.quando.compareTo(y.quando));
  return tutte.take(massimo).toList();
}
```

```dart
/// Giorni di calendario tra due date, ignorando le ore (immune all'ora legale).
int giorniTra(DateTime da, DateTime a) =>
    DateTime.utc(a.year, a.month, a.day).difference(DateTime.utc(da.year, da.month, da.day)).inDays;

int? giorniAllaScadenza(DateTime? scadenza, DateTime ora) => scadenza == null ? null : giorniTra(ora, scadenza.toLocal());

bool mostraAvvisoFirma(int? giorni) => giorni != null && giorni <= 3;

String testoAvvisoFirma(int giorni) {
  final quando = switch (giorni) { <= 0 => 'oggi', 1 => 'domani', _ => 'tra $giorni giorni' };
  return 'auri scade $quando: apri SideStore e tocca Aggiorna';
}

DateTime? quandoNotificaFirma(DateTime? scadenza, DateTime ora) {
  if (scadenza == null) return null;
  final s = scadenza.toLocal();
  final q = DateTime(s.year, s.month, s.day - 1, 9);
  return q.isAfter(ora) ? q : null;
}

bool mostraPromemoriaBackup({DateTime? ultimoBackup, required DateTime primoAvvio, required DateTime ora}) =>
    giorniTra(ultimoBackup ?? primoAvvio, ora) >= 7;

bool deveBloccare({required bool attivo, required DateTime? inBackgroundDa, required DateTime ora, required int minuti}) =>
    attivo && inBackgroundDa != null && ora.difference(inBackgroundDa).inMinutes >= minuti;

String nomeFileBackup(DateTime ora) =>
    'auri-backup-${ora.year}-${ora.month.toString().padLeft(2, '0')}-${ora.day.toString().padLeft(2, '0')}.auri';
```

```dart
/// Celle del mese con il lunedì come primo giorno; null = cella vuota.
List<DateTime?> celleMese(int anno, int mese) {
  final primo = DateTime(anno, mese, 1);
  final giorni = DateTime(anno, mese + 1, 0).day;
  final celle = <DateTime?>[
    for (var i = 1; i < primo.weekday; i++) null,
    for (var d = 1; d <= giorni; d++) DateTime(anno, mese, d),
  ];
  while (celle.length % 7 != 0) {
    celle.add(null);
  }
  return celle;
}
```

```dart
    this.nomeCorsoDiLaurea = '',
    this.cfuTotali = 180,

    this.bloccoAttivo = false,
    this.minutiRiblocco = 5,

  final int valoreLode;
  final bool bloccoAttivo;
  final int minutiRiblocco;
  final String preavvisi;
  final bool primoAvvioFatto;
  final DateTime? primoAvvioIl;
  final DateTime? ultimoBackup;

  static const kNome = 'nomeCorsoDiLaurea';
  static const kCfuTotali = 'cfuTotali';
  static const kLode = 'valoreLode';
  static const kBlocco = 'bloccoAttivo';
  static const kMinuti = 'minutiRiblocco';
  static const kPreavvisi = 'preavvisi';
  static const kPrimoAvvioFatto = 'primoAvvioFatto';
  static const kPrimoAvvioIl = 'primoAvvioIl';
  static const kUltimoBackup = 'ultimoBackup';

  factory Preferenze.daMappa(Map<String, String> m) {

      final v = int.tryParse(m[k] ?? '');
      return v != null && ok(v) ? v : def;

    return Preferenze(
      nomeCorsoDiLaurea: m[kNome] ?? '',
      cfuTotali: intero(kCfuTotali, 180, (v) => v > 0),
      valoreLode: intero(kLode, 30, (v) => v >= 30 && v <= 33),
      bloccoAttivo: m[kBlocco] == 'true',
      minutiRiblocco: intero(kMinuti, 5, (v) => v >= 0),
      preavvisi: m[kPreavvisi] ?? preavvisiPredefiniti,
      primoAvvioFatto: m[kPrimoAvvioFatto] == 'true',
      primoAvvioIl: DateTime.tryParse(m[kPrimoAvvioIl] ?? ''),
      ultimoBackup: DateTime.tryParse(m[kUltimoBackup] ?? ''),
    );
  }
```

```dart
import 'package:drift/drift.dart';

import '../domain/preferenze.dart';
import 'database.dart';

class CorsoConEsito {
  const CorsoConEsito(this.corso, this.esito);
  final Corso corso;
  final Esito? esito;
  bool get superato => esito != null;
}

class AppelloConCorso {
  const AppelloConCorso(this.appello, this.corso);
  final Appello appello;
  final Corso corso;
}

class Repository {
  Repository(this.db);
  final AuriDb db;

  Stream<List<CorsoConEsito>> guardaCorsi() {
    final q = db.select(db.corsi).join([leftOuterJoin(db.esiti, db.esiti.corsoId.equalsExp(db.corsi.id))])
      ..orderBy([OrderingTerm.asc(db.corsi.anno), OrderingTerm.asc(db.corsi.nome)]);
    return q.watch().map((righe) => [for (final r in righe) CorsoConEsito(r.readTable(db.corsi), r.readTableOrNull(db.esiti))]);
  }

  Future<int> salvaCorso(CorsiCompanion c) => db.into(db.corsi).insertOnConflictUpdate(c);

  Future<void> eliminaCorso(int id) => (db.delete(db.corsi)..where((t) => t.id.equals(id))).go();

  Future<void> registraEsito(EsitiCompanion e) => db.into(db.esiti).insertOnConflictUpdate(e);

  Future<void> eliminaEsito(int corsoId) => (db.delete(db.esiti)..where((t) => t.corsoId.equals(corsoId))).go();

  Stream<List<AppelloConCorso>> guardaAppelli() {
```

```dart
import 'dart:io';

import 'package:path/path.dart' as p;
import 'package:sqlite3/sqlite3.dart';

import '../data/database.dart';
import '../domain/avvisi.dart';

class BackupNonValido implements Exception {
  const BackupNonValido(this.motivo);
  final String motivo;
  @override
  String toString() => motivo;
}

Future<File> creaCopiaBackup(AuriDb db, Directory cartella, DateTime ora) async {
  final f = File(p.join(cartella.path, nomeFileBackup(ora)));
  if (f.existsSync()) f.deleteSync();
  await db.customStatement('VACUUM INTO ?', [f.path]);
  return f;
}

/// Controlla, in sola lettura, che il file sia un backup integro di auri.
void verificaBackup(File f) {
  Database? d;
  try {
    d = sqlite3.open(f.path, mode: OpenMode.readOnly);
    final integro = d.select('PRAGMA integrity_check').first.values.first;
    if (integro != 'ok') throw const BackupNonValido('Il file di backup è danneggiato.');
    final versione = d.select('PRAGMA user_version').first.values.first as int;
    final tabelle = d.select("SELECT name FROM sqlite_master WHERE type='table'").map((r) => r['name']).toSet();
    if (versione < 1 || !tabelle.containsAll(['corsi', 'esiti', 'appelli', 'impostazioni'])) {
      throw const BackupNonValido('Questo file non è un backup di auri.');
    }
  } on BackupNonValido {
    rethrow;
  } catch (_) {
    throw const BackupNonValido('Questo file non è un backup di auri.');
  } finally {
    d?.dispose();
  }
}

/// Va chiamata con il database dell'app CHIUSO.
Future<void> sostituisciDatabase({required File backup, required File destinazione}) async {
  verificaBackup(backup);
  await backup.copy(destinazione.path);
  for (final suffisso in ['-wal', '-shm', '-journal']) {
    final extra = File('${destinazione.path}$suffisso');
    if (extra.existsSync()) extra.deleteSync();
  }
}
```

```dart
import 'package:flutter_local_notifications/flutter_local_notifications.dart';
import 'package:flutter_timezone/flutter_timezone.dart';
import 'package:timezone/data/latest.dart' as tzdata;
import 'package:timezone/timezone.dart' as tz;

import '../domain/notifiche_piano.dart';

class NotificheService {
  final _plugin = FlutterLocalNotificationsPlugin();
  bool _pronta = false;

  IOSFlutterLocalNotificationsPlugin? get _ios => _plugin.resolvePlatformSpecificImplementation<IOSFlutterLocalNotificationsPlugin>();

  Future<void> inizializza() async {
    if (!Platform.isIOS) return;
    tzdata.initializeTimeZones();
    final zona = await FlutterTimezone.getLocalTimezone();
    tz.setLocalLocation(tz.getLocation(zona.identifier));
    await _plugin.initialize(const InitializationSettings(
      iOS: DarwinInitializationSettings(requestAlertPermission: false, requestBadgePermission: false, requestSoundPermission: false),
    ));
    _pronta = true;
  }

  Future<bool> chiediPermesso() async => !_pronta ? true : (await _ios?.requestPermissions(alert: true, sound: true) ?? false);

  Future<bool?> permessoConcesso() async {
    if (!_pronta) return null;
    final s = await _ios?.checkPermissions();
    return s?.isEnabled;
  }

  Future<void> riprogramma(List<NotificaPianificata> notifiche) async {
    if (!_pronta) return;
    await _plugin.cancelAll();
    for (final n in notifiche) {
      await _plugin.zonedSchedule(
        n.id,
        n.titolo,
        n.testo,
        tz.TZDateTime.from(n.quando, tz.local),
        const NotificationDetails(iOS: DarwinNotificationDetails()),
        androidScheduleMode: AndroidScheduleMode.inexactAllowWhileIdle,
      );
    }
  }
}
```

```dart
import 'dart:io';
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';
```

```dart
import '../../app/providers.dart';
import '../../data/apertura.dart';
import '../../domain/preferenze.dart';
import '../../services/backup.dart';

Future<void> esportaBackup(BuildContext context, WidgetRef ref, Rect origine) async {
  final db = ref.read(dbProvider).requireValue;
  final file = await creaCopiaBackup(db, await getTemporaryDirectory(), DateTime.now());
  final esito = await SharePlus.instance.share(ShareParams(files: [XFile(file.path)], sharePositionOrigin: origine));
  if (esito.status != ShareResultStatus.dismissed) {
    await ref.read(repositoryProvider).salvaPreferenza(Preferenze.kUltimoBackup, DateTime.now().toIso8601String());
  }
}

Future<void> ripristinaBackup(BuildContext context, WidgetRef ref) async {
  final scelta = await FilePicker.platform.pickFiles();
  final percorso = scelta?.files.single.path;
  if (percorso == null || !context.mounted) return;
  final ok = await showDialog<bool>(
    context: context,
    builder: (c) => AlertDialog(
      title: const Text('Ripristinare il backup?'),
      content: const Text('Sostituirà tutti i dati attuali.'),
      actions: [
        TextButton(onPressed: () => Navigator.pop(c, false), child: const Text('Annulla')),
        FilledButton(onPressed: () => Navigator.pop(c, true), child: const Text('Ripristina')),
      ],
    ),
  );
  if (ok != true) return;
  final backup = File(percorso);
  try {
    verificaBackup(backup);
  } on BackupNonValido catch (e) {
    if (context.mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(e.motivo)));
    return;
  }
  await ref.read(dbProvider).requireValue.close();
  await sostituisciDatabase(backup: backup, destinazione: await fileDatabase());
  ref.invalidate(dbProvider);
  if (context.mounted) ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('Backup ripristinato')));
}
```

```dart
import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

import '../../app/providers.dart';
import '../../app/tema.dart';
import '../../domain/preferenze.dart';

class PrimoAvvioPage extends ConsumerStatefulWidget {
  const PrimoAvvioPage({super.key});
  @override
  ConsumerState<PrimoAvvioPage> createState() => _PrimoAvvioPageState();
}

class _PrimoAvvioPageState extends ConsumerState<PrimoAvvioPage> {
  final _pagine = PageController();
  final _nome = TextEditingController();
  int _passo = 0;
  int _cfu = 180;
  int _lode = 30;
  bool _blocco = false;

  Future<void> _fine({bool salta = false}) async {
    final repo = ref.read(repositoryProvider);
    if (!salta) {
      await repo.salvaPreferenza(Preferenze.kNome, _nome.text.trim());
      await repo.salvaPreferenza(Preferenze.kCfuTotali, '$_cfu');
      await repo.salvaPreferenza(Preferenze.kLode, '$_lode');
      await repo.salvaPreferenza(Preferenze.kBlocco, '$_blocco');
    }
    await repo.salvaPreferenza(Preferenze.kPrimoAvvioIl, DateTime.now().toIso8601String());
    await repo.salvaPreferenza(Preferenze.kPrimoAvvioFatto, 'true');
  }

  void _avanti() {
    if (_passo == 2) {
      _fine();
      return;
    }
    setState(() => _passo++);
    _pagine.animateToPage(_passo, duration: const Duration(milliseconds: 260), curve: Curves.easeOutCubic);
  }

  @override
  Widget build(BuildContext context) {
    final touchId = ref.watch(touchIdDisponibileProvider).value ?? false;
    Widget passo(String t, Widget contenuto) => Column(mainAxisAlignment: MainAxisAlignment.center, crossAxisAlignment: CrossAxisAlignment.start, children: [
          Text(t, style: titolo(40)),
          const SizedBox(height: 24),
          contenuto,
        ]);
    return ColoredBox(
      color: AuriColori.sfondo,
      child: Center(
        child: SizedBox(
          width: 560,
          child: Column(children: [
            Expanded(
              child: PageView(controller: _pagine, physics: const NeverScrollableScrollPhysics(), children: [
                passo('Benvenuta in auri', Column(children: [
                  TextField(controller: _nome, decoration: const InputDecoration(labelText: 'Il tuo corso di laurea (es. Ingegneria Informatica)')),
                  const SizedBox(height: 16),
                  Row(children: [
                    const Text('CFU totali'),
                    const Spacer(),
                    for (final v in [120, 180, 300, 360])
                      Padding(padding: const EdgeInsets.only(left: 8), child: ChoiceChip(label: Text('$v'), selected: _cfu == v, onSelected: (_) => setState(() => _cfu = v))),
                  ]),
                ])),
                passo('Quanto vale la lode?', SegmentedButton<int>(
                  segments: [for (final v in [30, 31, 32, 33]) ButtonSegment(value: v, label: Text('$v'))],
                  selected: {_lode},
                  onSelectionChanged: (s) => setState(() => _lode = s.first),
                )),
                passo('Proteggere auri con Touch ID?', SwitchListTile(
                  contentPadding: EdgeInsets.zero,
                  title: const Text('Blocca con Touch ID'),
                  subtitle: Text(touchId ? 'Potrai cambiarlo nelle impostazioni' : 'Touch ID non è configurato: puoi attivarlo più tardi'),
                  value: _blocco && touchId,
                  onChanged: touchId ? (v) => setState(() => _blocco = v) : null,
                )),
              ]),
            ),
            Padding(
              padding: const EdgeInsets.only(bottom: 40),
              child: Row(children: [
                TextButton(onPressed: () => _fine(salta: true), child: const Text('Salta')),
                const Spacer(),
                FilledButton(onPressed: _avanti, child: Text(_passo == 2 ? 'Inizia' : 'Avanti')),
              ]),
            ),
          ]),
        ),
      ),
    );
  }
}
```
